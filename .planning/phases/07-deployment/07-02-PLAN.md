"""FastAPI application for SAR image compression."""
from contextlib import asynccontextmanager
from fastapi import FastAPI
from .endpoints import router

# Global model storage (populated at startup)
compressors = {}

@asynccontextmanager
async def lifespan(app: FastAPI):
    """Load models on startup, cleanup on shutdown."""
    from src.inference import SARCompressor
    import os

    # Find checkpoints
    checkpoint_dir = os.environ.get(
        "SARCODEC_CHECKPOINT_DIR",
        "notebooks/checkpoints"
    )

    # Load available models
    model_configs = {
        "4x": "resnet_c64_b64_cr4x_*/best.pth",
        "8x": "resnet_c32_b64_cr8x_*/best.pth",
        "16x": "resnet_c16_b64_cr16x_*/best.pth",
    }

    for name, pattern in model_configs.items():
        # Find matching checkpoint
        import glob
        matches = glob.glob(os.path.join(checkpoint_dir, pattern))
        if matches:
            try:
                compressors[name] = SARCompressor(matches[0])
                print(f"Loaded model: {name} from {matches[0]}")
            except Exception as e:
                print(f"Failed to load {name}: {e}")

    if not compressors:
        print("WARNING: No models loaded!")

    yield

    # Cleanup
    compressors.clear()

app = FastAPI(
    title="SAR Codec API",
    description="SAR image compression using learned autoencoders",
    version="1.0.0",
    lifespan=lifespan,
)

app.include_router(router)

# Allow running directly
if __name__ == "__main__":
    import uvicorn
    uvicorn.run(app, host="0.0.0.0", port=8000)
```

1. `GET /health`:
   - Returns HealthResponse with loaded models, device info, version
   - No authentication required

2. `POST /encode`:
   - Accepts: UploadFile (GeoTIFF), model: str query param (default "8x")
   - Returns: StreamingResponse with NPZ binary (latent + metadata)
   - Headers: Content-Disposition with filename
   - Validates file extension (.tif, .tiff)
   - Uses SARCompressor.compress() internally

3. `POST /decode`:
   - Accepts: UploadFile (NPZ), model: str query param
   - Returns: StreamingResponse with GeoTIFF binary
   - Uses SARCompressor.decompress() internally
   - Reconstructs GeoMetadata from NPZ

4. `POST /compress`:
   - Full pipeline: raw GeoTIFF -> compressed NPZ
   - Same as /encode (alias for user convenience)

5. `POST /decompress`:
   - Full pipeline: compressed NPZ -> GeoTIFF
   - Same as /decode (alias for user convenience)

@router.post("/encode")
async def encode_image(
    file: UploadFile = File(..., description="GeoTIFF image to compress"),
    model: str = Query("8x", enum=["4x", "8x", "16x"], description="Model variant"),
):
    """Encode a GeoTIFF to latent representation."""
    # Validate model exists
    if model not in compressors:
        raise HTTPException(404, f"Model '{model}' not loaded")

    # Validate file type
    if not file.filename.lower().endswith(('.tif', '.tiff')):
        raise HTTPException(400, "File must be GeoTIFF (.tif or .tiff)")

    compressor = compressors[model]
    start_time = time.time()

    try:
        # Read file content
        content = await file.read()

        # Read GeoTIFF from memory
        from rasterio.io import MemoryFile
        with MemoryFile(content) as memfile:
            with memfile.open() as src:
                data = src.read(1).astype(np.float32)
                geo_metadata = extract_geo_metadata(src)

        # Compress
        latent, tile_metadata = compressor.compress(data)

        # Build response NPZ
        buffer = io.BytesIO()
        np.savez_compressed(
            buffer,
            latent=latent,
            metadata=json.dumps({
                "geo_metadata": serialize_geo_metadata(geo_metadata),
                "tile_metadata": tile_metadata,
            })
        )
        buffer.seek(0)

        processing_time = time.time() - start_time

        return StreamingResponse(
            buffer,
            media_type="application/octet-stream",
            headers={
                "Content-Disposition": f"attachment; filename={file.filename}.npz",
                "X-Processing-Time-Ms": str(int(processing_time * 1000)),
            }
        )
    except Exception as e:
        raise HTTPException(500, f"Compression failed: {str(e)}")
```
